

### 1\. Binary Search



#include <stdio.h>
int main(){
    int a[5]={10,20,30,40,50};
    int key=30,low=0,high=4,mid,found=0;

    while(low<=high){
        mid=(low+high)/2;
        if(a[mid]==key){
            found=1;
            break;
        }
        else if(a[mid]<key)
            low=mid+1;
        else
            high=mid-1;
    }

    if(found)
        printf("Found at position %d",mid+1);
    else
        printf("Not Found");

    return 0;
}


### 2\. Quick Sort



#include <stdio.h>
void quickSort(int a[],int low,int high){
    int i,j,pivot,temp;
    if(low<high){
        pivot=a[high];
        i=low-1;
        for(j=low;j<high;j++){
            if(a[j]<pivot){
                i++;
                temp=a[i];
                a[i]=a[j];
                a[j]=temp;
            }
        }
        temp=a[i+1];
        a[i+1]=a[high];
        a[high]=temp;

        int p=i+1;
        quickSort(a,low,p-1);
        quickSort(a,p+1,high);
    }
}
int main(){
    int a[5]={5,2,8,1,3};
    quickSort(a,0,4);
    for(int i=0;i<5;i++)
        printf("%d ",a[i]);
    return 0;
}




### 3\. Insertion Sort



#include <stdio.h>
int main(){
    int a[5]={5,3,4,1,2};
    int i,j,key;
    for(i=1;i<5;i++){
        key=a[i];
        j=i-1;
        while(j>=0&&a[j]>key){
            a[j+1]=a[j];
            j--;
        }
        a[j+1]=key;
    }
    for(i=0;i<5;i++)
        printf("%d ",a[i]);
    return 0;
}




### 4\. Selection Sort


#include <stdio.h>
int main(){
    int a[5]={5,3,4,1,2};
    int i,j,min,temp;
    for(i=0;i<5;i++){
        min=i;
        for(j=i+1;j<5;j++){
            if(a[j]<a[min])
                min=j;
        }
        temp=a[i];
        a[i]=a[min];
        a[min]=temp;
    }
    for(i=0;i<5;i++)
        printf("%d ",a[i]);
    return 0;
}
